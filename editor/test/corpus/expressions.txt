function check_val(x: i32): i32 {
    let y = if (x > 0) {
        return x;
    } else {
        return 0;
    };
    return y;
}

function demo(opt: Option(i32)): i32 {
    let f = (x: i32) => x + 1;
    return match (opt) {
        Some(v) => f(v),
        None => 0,
    };
}
